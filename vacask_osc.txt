// Circuit
// -------
// A single-node LC oscillator with a Van der Pol nonlinear conductance
//
//   C  = 1 nF  (ddt)
//   L  = 1 uH  (idt, IL(0)=0)
//   VdP: I(V) = (-G0 + G1*V^2)*V
//        G0 = 0.01 S, G1 = 0.01 S/V^2  => amplitude ~ sqrt(4*G0/(3*G1)) ~ 1.15 V
//
// Resonant frequency:
//   f0 = 1 / (2*pi*sqrt(L*C)) = 1 / (2*pi*sqrt(1e-6*1e-9)) ~= 5.033 MHz
//   T0 ~= 198.69 ns
//
// To kick the oscillation, ic={"1", 0.9} sets V(1)=0.9 V at t=0.
//
// PSS parameters
// --------------
//   Tper  = 200ns  (initial guess, ~1% above exact T0)
//   Tstab = 5us    (25 periods; enough for the oscillation to settle)

ground 0

`include "constants.vams"
`include "disciplines.vams"

// Full LC-VdP oscillator in one module.
// Implements:
//   - Capacitor: I <+ ddt(C * V)
//   - Inductor:  I <+ idt(V/L, 0)   [IL(0)=0]
//   - VdP conductance: I <+ (-G0 + G1*V^2)*V
// Hardcoded values (VACASK Verilog-A parameters are not yet settable from
// the netlist for embedded compiled models):
//   C=1e-9 F, L=1e-6 H, G0=0.01 S, G1=0.01 S/V^2
module vdposc(A, B);
    inout A, B;
    electrical A, B;

    analog begin
        // Capacitor
        I(A,B) <+ ddt(1e-9 * V(A,B));
        // Inductor (IL(0)=0)
        I(A,B) <+ idt(V(A,B) / 1e-6, 0);
        // Van der Pol nonlinear conductance
        I(A,B) <+ (-0.01 + 0.01 * pow(V(A,B), 2)) * V(A,B);
    end
endmodule

//
// Tank: L=10 nH, C1=C2=6.3 pF -> C_eff=3.15 pF
